def esc(s):
    return html.escape('' if s is None else str(s), quote=True)


def _para(text):
    """Turn a record's prose into paragraphs without inventing structure."""
    if not text:
        return ''
    blocks = [b.strip() for b in re.split(r'\n\s*\n', str(text)) if b.strip()]
    return ''.join('<p>%s</p>' % esc(b).replace('\n', '<br>') for b in blocks)


def _chip(verdict):
    v = (verdict or 'OPEN').upper()
    cls = {'SUPPORTED': 'v-sup', 'REFUTED': 'v-ref', 'INCONCLUSIVE': 'v-inc'}.get(v, 'v-open')
    return '<span class="chip %s">%s</span>' % (cls, esc(v))

def _is_label(part):
    """A section heading written inline, not a claim.

    Recognised by shape rather than by wording: a short run whose letters are almost all capitals.
    The records use "MECHANISM.", "WHAT THIS DOES NOT CLOSE," and numbered all-caps clause titles;
    matching those strings would work today and silently stop working on the next record that
    invents a heading, so the test is on the form.
    """
    p = part.strip().lstrip('(').lstrip('0123456789').lstrip(') ').strip()
    letters = [c for c in p if c.isalpha()]
    if not letters:
        return False
    upper = sum(1 for c in letters if c.isupper()) / float(len(letters))
    return upper >= 0.8 and len(p) <= 120


def _sentence(text, prefer_digit=False, cap=230):
    """First sentence of a field, optionally the first one carrying a number.

    Condensation by selection. The cell shows a sentence the record already contains; the
    expansion shows the field entire. Nothing is rewritten, so a cell can be wrong only in the
    way its source is wrong.
    """
    if not text:
        return ''
    s = ' '.join(str(text).split())
    parts = [p for p in re.split(r'(?<=[.!?])\s+(?=[A-Z0-9(])', s) if p.strip()]
    body = [p for p in parts if not _is_label(p)] or parts
    pick = None
    if prefer_digit:
        for p in body:
            if re.search(r'\d', p):
                pick = p
                break
    if pick is None:
        pick = body[0] if body else s
    if len(pick) > cap:
        pick = pick[:cap].rsplit(' ', 1)[0] + '\u2026'
    return pick


def humanise(hid):
    """A hypothesis id is a sentence with hyphens. Restore the sentence; drop the date stamp."""
    s = re.sub(r'-?20\d{6}([tT]\d{4}[zZ])?$', '', str(hid or ''))
    s = s.replace('-', ' ').replace('_', ' ').strip()
    return (s[:1].upper() + s[1:]) if s else str(hid)

def measure_block(m, prev_value):
    """Latest comparable measurement against its target, and the change since session start.

    An unknown stays unknown: when the receipt is absent, or when no earlier comparable value
    exists to difference against, the card says so instead of showing a zero. A zero is a
    measurement and would assert that the quantity was observed and found unchanged.
    """
    if m.get('status') != 'READ':
        return ('<div class="measure"><div class="mcard"><span class="label">Measured position</span>'
                '<span class="big">Unknown</span><p class="sub">%s</p></div></div>'
                % esc(m.get('why') or 'no adjudication receipt is present'))
    v, target = float(m['value']), float(m['target'])
    unit = m.get('units') or ''
    frac = max(0.0, min(1.0, v / target)) if target else 0.0
    if prev_value is None:
        change = ('<span class="big">Unknown</span>'
                  '<p class="sub">No earlier comparable measurement is recorded inside this '
                  'session, so the change across it cannot be stated. Unknown is not zero.</p>')
        cls = ' flat'
    else:
        d = v - float(prev_value)
        if abs(d) < 1e-9:
            change = ('<span class="big">No change</span>'
                      '<p class="sub">The same adjudicated measurement stands at the start and the '
                      'end of this session. Work in the table below is diagnostic; none of it '
                      'produced a new governed number.</p>')
            cls = ' flat'
        else:
            change = ('<span class="big">%+,.0f<span class="unit"> %s</span></span>'
                      '<p class="sub">Against %s at the start of this session.</p>'
                      % (d, esc(unit), format(float(prev_value), ',.0f')))
            cls = ''
    return (
        '<div class="measure">'
        '<div class="mcard"><span class="label">Measured &mdash; %s</span>'
        '<span class="big">%s<span class="unit"> %s</span></span>'
        '<div class="bar"><i style="width:%.1f%%"></i></div>'
        '<p class="barnote">%.1f%% of the %s target</p>'
        '<p class="sub">%s over %s updates and %s seconds, adjudicated against the target by the '
        'same statistic.</p></div>'
        '<div class="mcard"><span class="label">Target</span>'
        '<span class="big">%s<span class="unit"> %s</span></span>'
        '<p class="sub">%s %s. The target is the condition, not a forecast; this page carries no '
        'estimate of when it will be reached.</p></div>'
        '<div class="mcard%s"><span class="label">Change since session start</span>%s</div>'
        '</div>'
        % (esc(m.get('label') or 'position'), format(v, ',.0f'), esc(unit),
           frac * 100.0, frac * 100.0, format(target, ',.0f'),
           esc(str(m.get('statistic') or 'the declared statistic')),
           format(int(m.get('updates') or 0), ','),
           format(float(m.get('seconds') or 0.0), ',.0f'),
           format(target, ',.0f'), esc(unit),
           esc(str(m.get('comparator') or 'compared')), esc(str(m.get('statistic') or '')),
           cls, change))

def _kind_of(h, rows):
    """Diagnostic or governed, decided by the record rather than by tone.

    A result is GOVERNED only if a registry row says it moves an acceptance gate. Everything
    else is diagnostic: it measures where time goes, under instrumentation that changes the
    timing, and earns no progress credit.
    """
    for r in rows:
        mg = str(r.get('moves_gate') or '').strip().lower()
        if mg and not mg.startswith('no'):
            return 'governed'
    ran_on_card = any(e.get('gpu') for e in h.get('events', []))
    return 'diagnostic on the card' if ran_on_card else 'diagnostic, existing capture'

def figure_svg(sweep, audit):
    """Two layouts of one chart: wide with the key beside the plot, narrow with it below.

    The page shows one or the other by media query. Neither is a summary of the other --
    both carry every point and every kernel row, so a phone reader is not given less.
    """
    return ('<div class="fig-wide">%s</div><div class="fig-narrow">%s</div>'
            % (_figure_svg(sweep, audit, WIDE_FIG), _figure_svg(sweep, audit, NARROW_FIG)))


# Geometry only. W/H are the viewBox; L/R/T/B are the plot's margins inside it; when
# key_below is set the kernel key is laid out under the plot from key_y instead of in the
# right margin, which is why the narrow layout can give R back to the data.
WIDE_FIG = dict(W=940, H=430, L=62, R=250, T=26, B=68,
                key_below=False, key_y=0, key_x=None, title_wrap=False)
# R=22 holds the right half of the "16384" tick, which is centred on the plot's right edge.
# W=362 is what the content needs once the key sits below, and it is close enough to a phone
# column that the type scales to about 0.96 rather than 0.4.
NARROW_FIG = dict(W=362, H=648, L=44, R=22, T=40, B=378,
                  key_below=True, key_y=336, key_x=8, title_wrap=True)


def _figure_svg(sweep, audit, g):
    srows = sweep['rows']
    arows = audit['rows']
    aud = {}
    for a in arows:
        aud.setdefault(a['m'], {})[a['arm']] = a

    W, H = g['W'], g['H']
    L, R, T, B = g['L'], g['R'], g['T'], g['B']
    xs = [r['m'] for r in srows]
    x0, x1 = math.log(min(xs)), math.log(max(xs))
    y0, y1 = 0.90, 1.80

    def px(m):
        return L + (math.log(m) - x0) / (x1 - x0) * (W - L - R)

    def py(v):
        return T + (y1 - v) / (y1 - y0) * (H - T - B)

    axis_y = py(y0)
    p = ['<svg viewBox="0 0 %d %d" role="img" xmlns="http://www.w3.org/2000/svg" '
         'aria-label="Measured speedup of the 16-bit compute-type request over the 32-bit request, '
         'against m, with kernel identity marked only at the five shapes the audit measured.">' % (W, H),
         '<style>'
         '.ax{stroke:var(--rule);stroke-width:1}.gr{stroke:var(--rule-2);stroke-width:1}'
         '.tk{fill:var(--ink-3);font:11px "IBM Plex Sans",sans-serif}'
         '.tkm{fill:var(--ink-3);font:11px "IBM Plex Mono",monospace}'
         '.ln{fill:none;stroke:var(--ink-2);stroke-width:1.6}'
         '.lg{fill:var(--ink-2);font:11.5px "IBM Plex Sans",sans-serif}'
         '.lgb{fill:var(--ink);font:600 11.5px "IBM Plex Sans",sans-serif}'
         '.kn{fill:var(--amber);font:600 10px "IBM Plex Mono",monospace}'
         '.dip{stroke:var(--amber);stroke-width:1.2;stroke-dasharray:3 3}'
         '.ttl{fill:var(--ink);font:600 12px "IBM Plex Sans",sans-serif}'
         '</style>']
    # gridlines + y ticks
    for v in (1.0, 1.2, 1.4, 1.6, 1.8):
        p.append('<line class="gr" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>' % (L, py(v), W - R, py(v)))
        p.append('<text class="tkm" x="%.1f" y="%.1f" text-anchor="end">%.1fx</text>' % (L - 8, py(v) + 4, v))
    p.append('<line class="ax" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>' % (L, axis_y, W - R, axis_y))
    p.append('<line class="ax" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>' % (L, T, L, axis_y))
    # parity line
    p.append('<line class="ax" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f" stroke-dasharray="2 4"/>'
             % (L, py(1.0), W - R, py(1.0)))
    # x ticks
    for m in xs:
        p.append('<line class="ax" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>'
                 % (px(m), axis_y, px(m), axis_y + 4))
    for m in (512, 1024, 2048, 4096, 8192, 16384):
        p.append('<text class="tkm" x="%.1f" y="%.1f" text-anchor="middle">%d</text>'
                 % (px(m), axis_y + 18, m))
    p.append('<text class="tk" x="%.1f" y="%.1f" text-anchor="middle">m  (k = n = 1024, log scale)</text>'
             % ((L + W - R) / 2.0, axis_y + 38))
    if g['title_wrap']:
        p.append('<text class="ttl" x="%.1f" y="%.1f">speedup: 16-bit compute-type</text>'
                 % (L - 36, T - 22))
        p.append('<text class="ttl" x="%.1f" y="%.1f">request over 32-bit</text>'
                 % (L - 36, T - 8))
    else:
        p.append('<text class="ttl" x="%.1f" y="%.1f">speedup: 16-bit compute-type request '
                 'over 32-bit</text>' % (L, T - 8))
    # line
    p.append('<polyline class="ln" points="%s"/>' % ' '.join(
        '%.1f,%.1f' % (px(r['m']), py(r['speedup_16f_over_32f'])) for r in srows))
    # dip marker
    p.append('<line class="dip" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>'
             % (px(2048), T + 4, px(2048), axis_y))
    p.append('<text class="kn" x="%.1f" y="%.1f" text-anchor="middle">dip, unexplained</text>'
             % (px(2048), T + 1))
    # points
    for r in srows:
        m, v = r['m'], r['speedup_16f_over_32f']
        joined = m in aud and '32F' in aud[m] and '16F' in aud[m]
        if joined:
            p.append('<rect x="%.1f" y="%.1f" width="8" height="8" fill="var(--amber)"/>'
                     % (px(m) - 4, py(v) - 4))
        else:
            p.append('<circle cx="%.1f" cy="%.1f" r="4" fill="var(--paper)" '
                     'stroke="var(--ink-2)" stroke-width="1.5"/>' % (px(m), py(v)))
    # kernel key, only for joined shapes: right margin when wide, under the plot when narrow
    kx = g['key_x'] if g['key_below'] else W - R + 10
    y = g['key_y'] if g['key_below'] else T + 16
    p.append('<text class="lgb" x="%.1f" y="%.1f">Kernel actually selected</text>' % (kx, y))
    y += 13
    p.append('<text class="lg" x="%.1f" y="%.1f">(profiled audit; 5 shapes only)</text>' % (kx, y))
    y += 19
    for m in sorted(aud):
        if not (m in [r['m'] for r in srows]):
            continue
        a32, a16 = aud[m].get('32F'), aud[m].get('16F')
        if not (a32 and a16):
            continue
        p.append('<text class="lgb" x="%.1f" y="%.1f">m = %d</text>' % (kx, y, m))
        y += 14
        p.append('<text class="kn" x="%.1f" y="%.1f">32F tile %s</text>' % (kx, y, a32['tile_in_name']))
        y += 13
        p.append('<text class="kn" x="%.1f" y="%.1f">16F tile %s</text>' % (kx, y, a16['tile_in_name']))
        y += 18
    p.append('<rect x="%.1f" y="%.1f" width="8" height="8" fill="var(--amber)"/>' % (kx, y - 7))
    p.append('<text class="lg" x="%.1f" y="%.1f">kernel evidence</text>' % (kx + 14, y))
    y += 16
    p.append('<circle cx="%.1f" cy="%.1f" r="4" fill="var(--paper)" stroke="var(--ink-2)" '
             'stroke-width="1.5"/>' % (kx + 4, y - 4))
    p.append('<text class="lg" x="%.1f" y="%.1f">no kernel evidence</text>' % (kx + 14, y))
    p.append('</svg>')
    return ''.join(p)


def figure_table(sweep, audit):
    aud = {}
    for a in audit['rows']:
        aud.setdefault(a['m'], {})[a['arm']] = a
    sw = dict((r['m'], r) for r in sweep['rows'])
    allm = sorted(set(list(sw)) | set(list(aud)))
    tr = []
    for m in allm:
        s, a = sw.get(m), aud.get(m, {})
        a32, a16 = a.get('32F'), a.get('16F')
        if s:
            sp = '%.4f' % s['speedup_16f_over_32f']
            spread = '%.3f / %.3f' % (s['spread_32f'], s['spread_16f'])
        else:
            sp, spread = '&mdash;', '&mdash;'
        if a32 and a16:
            k32 = '<code>%s</code>' % esc(a32['tile_in_name'])
            k16 = '<code>%s</code>' % esc(a16['tile_in_name'])
            same = 'same' if a32['kernel'] == a16['kernel'] else '<strong>different</strong>'
            ar = '%.3f' % (a32['profiled_us_NOT_A_RATE'] / a16['profiled_us_NOT_A_RATE'])
        else:
            k32 = k16 = same = ar = '<span class="none">no audit row</span>'
        tr.append('<tr><td class="num">%d</td><td class="num">%s</td><td class="num">%s</td>'
                  '<td>%s</td><td>%s</td><td>%s</td><td class="num">%s</td>'
                  '<td>%s</td></tr>'
                  % (m, sp, spread, k32, k16, same, ar,
                     'sweep only' if s and not (a32 and a16) else
                     ('audit only' if a32 and a16 and not s else 'both')))
    return ('<div class="tablewrap"><table><thead><tr>'
            '<th>m</th><th>sweep speedup</th><th>spread 32F / 16F</th>'
            '<th>32F tile</th><th>16F tile</th><th>kernel</th>'
            '<th>audit ratio<br>(profiled, not a rate)</th><th>measured in</th>'
            '</tr></thead><tbody>%s</tbody></table></div>' % ''.join(tr))
