```python
class ArchitectureMapError(RuntimeError):
    def __init__(self, code: str, detail: str):
        super().__init__(f"{code}: {detail}")
        self.code = code
        self.detail = detail

def canonical_json(value: object) -> bytes:
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")

def classify_paths(paths: Sequence[str], policy: Mapping[str, object]) -> list[dict[str, object]]:
    """Return one lexically ordered row per tracked path; refuse zero or multiple matching rules."""
    rows = []
    for path in sorted(paths):
        matches = [item for item in policy["path_rules"] if rule_matches(item, path)]
        if len(matches) != 1:
            code = "UNCOVERED_PATH" if not matches else "OVERLAPPING_PATH_RULES"
            raise ArchitectureMapError(code, path)
        rows.append(classification_row(path, matches[0], policy))
    return rows
```

```python
class ColoringResult:
    k: int
    assignment: dict[str, int]
    k_minus_one_unsat: dict[str, object]

def solve_minimum_coloring(nodes: Sequence[str], edges: Sequence[tuple[str, str]], precedence: Sequence[tuple[str, str]], max_states: int) -> ColoringResult:
    graph = normalized_graph(nodes, edges, precedence)
    for k in range(1, len(graph.nodes) + 1):
        search = exact_k_color_search(graph, k, max_states=max_states)
        if search.witness is not None:

    raise ArchitectureMapError("COLORING_INTERNAL_ERROR", "finite graph produced no coloring")
```

Each RED row contains `leg_id`, `status`, `command_argv_sha256`, `started_at`, `stopped_at`, `returncode`, `stdout_sha256`, `stderr_sha256`, `failure_class`, and `claim_boundary`. `SKIP` requires a reason and may never satisfy a green predicate.
```python
def derive_self_sha256(receipt: Mapping[str, object]) -> str:
    payload = dict(receipt)
    payload.pop("self_sha256", None)
    return hashlib.sha256(canonical_json(payload)).hexdigest()

def write_no_overwrite_receipt(path: Path, receipt: Mapping[str, object]) -> tuple[str, str]:
    if path.exists():
        raise ArchitectureMapError("OUTPUT_EXISTS", str(path))
    payload = dict(receipt)
    payload["self_sha256"] = derive_self_sha256(payload)
    raw = canonical_json(payload) + b"\n"
    path.write_bytes(raw)
    return hashlib.sha256(raw).hexdigest(), str(payload["self_sha256"])

def verify_receipt(repo: Path, policy_path: Path, receipt_path: Path) -> dict[str, object]:
    supplied = json.loads(receipt_path.read_text(encoding="utf-8"))
    recomputed = compile_receipt(repo, load_policy(policy_path), supplied["red_matrix"])
    require_equal_terminal_fields(supplied, recomputed)
    verify_coloring_certificate(recomputed["conflict_graph"], supplied["coloring"], supplied["max_states"])
    return {"result": "PASS", "self_sha256": derive_self_sha256(supplied)}
```
